## Prerequisites

Before getting started, make sure you have:

- A running web application with user authentication in your chosen framework.

- A Forma account with your **Workspace ID** and **SDK Connection URL** (the `appUrl` in every snippet below), available under **Settings → Workspace → Web & Mobile SDK**, on the **Workspace details** card.

## HTML

All you need to do is copy a `<script>` tag to your HTML head:

```javascript
<!-- START Forma Surveys -->
<script type="text/javascript">
!function(){
    var appUrl = "https://app.forma.ylam.ai"; // use PUBLIC_URL if you are using multi-domain setup, otherwise use WEBAPP_URL
    var workspaceId = "<your-workspace-id>";
var t=document.createElement("script");t.type="text/javascript",t.async=!0,t.src=appUrl+"/js/forma.umd.cjs";t.onload=function(){window.forma.setup({workspaceId: workspaceId, appUrl: appUrl})};var e=document.getElementsByTagName("script")[0];e.parentNode.insertBefore(t,e)}();
</script>
<!-- END Forma Surveys -->
```

| Name           | Type   | Description                            |
| -------------- | ------ | -------------------------------------- |
| workspaceId    | string | Forma Workspace ID.               |
| appUrl         | string | URL of the hosted Forma instance. |

Now, visit the [Validate Your Setup](#validate-your-setup) section to verify your setup!

## Next.js

Next.js apps use either the **App Directory** or the **Pages Directory**. Since the Forma SDK runs on the client side, follow these steps based on your setup:

- **App Directory**: Create a new component in `app/forma.tsx` and call it in `app/layout.tsx`.

- **Pages Directory**: Initialize Forma directly in `_app.tsx`.

Code snippets for the integration for both conventions are provided to further assist you.

```bash npm

```


```bash yarn

```

### App directory

```typescript app/forma.tsx
"use client";

import { usePathname, useSearchParams } from "next/navigation";
import { useEffect } from "react";
import forma from "@forma/js";

export default function FormaProvider() {
  const pathname = usePathname();
  const searchParams = useSearchParams();

  useEffect(() => {
    forma.setup({
      workspaceId: "<workspace-id>",
      appUrl: "<app-url>", // use PUBLIC_URL if you are using multi-domain setup, otherwise use WEBAPP_URL
    });
  }, []);

  useEffect(() => {
    forma?.registerRouteChange();
  }, [pathname, searchParams]);

  return null;
}
```

```typescript app/layout.tsx
// other imports
import FormaProvider from "./forma";
import { Suspense } from "react";

export default function RootLayout({
  children,
}: {
  children: React.ReactNode;
}) {
  return (
    <html lang="en">
      <Suspense>
        <FormaProvider />
      </Suspense>
      <body>{children}</body>
    </html>
  );
}
```

### Pages directory

```javascript src/pages/_app.tsx
// other import
import { useRouter } from "next/router";
import { useEffect } from "react";
import forma from "@forma/js";

if (typeof window !== "undefined") {
  forma.setup({
    workspaceId: "<workspace-id>",
    appUrl: "<app-url>", // use PUBLIC_URL if you are using multi-domain setup, otherwise use WEBAPP_URL
  });
}

export default function App({ Component, pageProps }: AppProps) {
  const router = useRouter();

  useEffect(() => {
    // Connect next.js router to Forma
    const handleRouteChange = forma?.registerRouteChange;
    router.events.on("routeChangeComplete", handleRouteChange);

    return () => {
      router.events.off("routeChangeComplete", handleRouteChange);
    };
  }, []);
  return <Component {...pageProps} />;
}
```

Install the Forma iOS SDK using the following steps:

**Swift Package Manager**

1. In Xcode choose **File → Add Packages…**
2. Enter your repo URL (e.g. `https://github.com/yldm-tech/ios.git`)
3. Choose version rule (e.g. "Up to Next Major" starting at `2.1.0`).
4. Import in your code:

   ```swift
   import FormaSDK
   ```

**CocoaPods**

1. Add the following to your `Podfile`:

   ```ruby
   platform :ios, '16.4'
   use_frameworks! :linkage => :static

   target 'YourTargetName' do
     pod 'FormaSDK', '~> 2.1'
   end
   ```

2. Run `pod install` in your app directory
3. Import in your code:
   ```swift
   import FormaSDK
   ```

Now start using FormaSDK

```swift
import FormaSDK

// 1. Build your config (you can also inject userId + attributes here)
let config = FormaConfig.Builder(
        appUrl: "https://your‑app.bricks.com", // use PUBLIC_URL if you are using multi-domain setup, otherwise use WEBAPP_URL
        workspaceId: "YOUR_WORKSPACE_ID"
    )
    .setLogLevel(.debug)
    .build()

// 2. Initialize the SDK (once per launch)
Forma.setup(with: config)

// 3. Identify the user
Forma.setUserId("user‑123")

// 4. Track events
Forma.track("button_pressed")

// 5. Set or add user attributes
Forma.setAttribute("blue", forKey: "favoriteColor")
Forma.setAttributes([
    "plan": "pro",
    "tier": "gold"
])

// 6. Change language (no userId required):
Forma.setLanguage("de")

// 7. Attach Embedded Data to future responses
Forma.setEmbeddedData(["screen": "checkout", "plan": "pro"])

// 8. Log out (no userId required):
Forma.logout()

// 9. Clean up SDK state (optional):
Forma.cleanup(waitForOperations: true) {
    print("SDK torn down")
}
```

<Note>
  `setEmbeddedData` attaches context to whichever survey is displayed next, without tying it to a trigger.
  See [Mobile SDKs](/surveys/embedded-data/mobile-sdks) for the value types and for the `nil` caveat that is
  specific to Swift.
</Note>

### Usage

Mount the `Forma` widget once, high in your widget tree (e.g. alongside your app's home). It initializes the SDK on mount and renders any triggered survey in a modal overlay:

```dart
import 'package:flutter/material.dart';
import 'package:forma/forma.dart';

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      home: Stack(
        children: [
          const HomePage(),
          // Initializes the SDK and hosts the survey overlay.
          Forma(
            appUrl: "<app-url>", // use PUBLIC_URL if you are using multi-domain setup, otherwise use WEBAPP_URL
            workspaceId: "<workspace-id>",
          ),
        ],
      ),
    );
  }
}
```

Then drive the SDK through its static API from anywhere:

```dart
// 1. Trigger a code action — shows a matching, eligible survey.
await Forma.track("button_clicked");

// 2. Identify the current user.
await Forma.setUserId("user_123");

// 3. Set or add user attributes.
await Forma.setAttribute("plan", "pro");
await Forma.setAttributes({"plan": "pro", "mrr": 99, "signup_date": DateTime.now()});

// 4. Set the survey language.
await Forma.setLanguage("de");

// 5. Attach Embedded Data to future responses (synchronous, no queue).
Forma.setEmbeddedData({"screen": "checkout", "plan": "pro"});

// 6. Reset to an anonymous session on sign-out.
await Forma.logout();
```

<Note>
  Prefer to initialize before any UI mounts? Call `Forma.setup(appUrl: ..., workspaceId: ...)` directly — the widget still needs to be in the tree to render surveys, but it won't re-run setup (setup is idempotent). `setup`, `track`, `setUserId`, `setAttribute`, `setAttributes`, `setLanguage` and `logout` are sequenced through an internal command queue, so they run in submission order and return a `Future<Result<void, FormaError>>`. `setEmbeddedData` and `clearEmbeddedData` are the exceptions: they are synchronous and return `void`, so a context push at launch is not dropped while setup is still running.
</Note>

### Public API

| Method                                              | Description                                                              |
| --------------------------------------------------- | ------------------------------------------------------------------------ |
| `Forma(appUrl:, workspaceId:, logLevel:)`      | Host widget — initializes the SDK and renders triggered surveys.         |
| `Forma.setup({appUrl, workspaceId, logLevel})` | Imperative initialization. Idempotent.                                   |
| `Forma.track(String name)`                     | Fire a code action; shows a matching eligible survey.                    |
| `Forma.setUserId(String userId)`               | Identify the contact. Switching ids resets the prior user's state first. |
| `Forma.setAttribute(String key, Object value)` | Set one attribute (`String` / `num` / `DateTime`).                       |
| `Forma.setAttributes(Map<String, Object>)`     | Set several attributes at once.                                          |
| `Forma.setLanguage(String language)`           | Set the survey language.                                                 |
| `Forma.setEmbeddedData(Map<String, Object?>)`  | Attach [Embedded Data](/surveys/embedded-data/mobile-sdks) to future responses. Synchronous. |
| `Forma.clearEmbeddedData('key')`               | Remove one Embedded Data key. With no argument, clears all of them.      |
| `Forma.logout()`                               | Reset to an anonymous session.                                           |

Attribute keys must be lowercase letters, numbers, and underscores, and start with a letter. `DateTime` values are sent as UTC ISO-8601 strings; numbers stay numbers. Identity and attribute changes are debounced (~500 ms) and coalesced into a single backend request.

## Content Security Policy

If your app serves a strict `Content-Security-Policy` header, the browser can block the Forma widget before anything renders. Three directives are involved.

<Note>
  This applies to the JavaScript SDK only. Link surveys are served by Forma itself, and the mobile
  SDKs render surveys in a WebView, so neither is affected by your app's CSP.
</Note>

### Allowing scripts and API calls

The SDK loads two scripts from your Forma instance: `forma.umd.cjs` (the SDK) and `surveys.umd.cjs` (the survey renderer, fetched the first time a survey is shown). It then talks to the client API on the same host. Both directives need to point at your `appUrl`:

```text
script-src 'self' https://app.forma.ylam.ai;
connect-src 'self' https://app.forma.ylam.ai;
```

The SDK does not put a nonce on the script tags it creates, so allowlisting the host is what makes them load. `'strict-dynamic'` works instead of a host allowlist, but only if the script that loads the SDK carries a matching nonce or hash of its own — both bundles are inserted programmatically, so they inherit trust from that root. Without a trusted root, `'strict-dynamic'` blocks the SDK before it ever reaches `surveys.umd.cjs`.

### Passing a nonce for survey styles

Surveys are styled by two `<style>` elements the renderer appends to `<head>`: `forma__css` for the base stylesheet and `forma__css__custom` for your workspace and survey theme. `style-src` treats an injected `<style>` element as inline CSS, so a policy without `'unsafe-inline'` blocks both and surveys appear unstyled.

Instead of loosening the policy, hand the SDK the nonce your server already generated for the request. `setNonce` stamps it onto both style elements, which satisfies `style-src 'nonce-...'`:

```javascript Setting the CSP nonce
forma.setNonce("<your-request-nonce>");

forma.setup({
  workspaceId: "<workspace-id>",
  appUrl: "<app-url>",
});
```

The matching header side, with the nonce in both style directives:

```text
style-src 'self' 'nonce-<your-request-nonce>';
style-src-elem 'self' 'nonce-<your-request-nonce>';
```

`style-src-elem` is what the browser consults for `<style>` elements whenever it is present, and it does not fall back to `style-src` — so a nonce listed only under `style-src` leaves surveys unstyled on any policy that sets both. If your policy has no `style-src-elem`, the first line alone is enough.

Pass the raw value only — not the `nonce-` prefix that appears in the header. Passing `undefined` stops the SDK nonce-ing style elements it creates from then on; elements already in the page keep the attribute they were given.

<Note>
  **Note**: call `setNonce` before `setup`. The SDK stores the nonce and applies it when the survey
  renderer loads, so setting it early covers every survey in the page's lifetime. A nonce must be
  regenerated per response and match the one in the header, so read it from wherever your framework
  exposes it rather than hardcoding a value.
</Note>

## Debugging Forma Integration

<Note>
  The debug mode is only available in the JavaScript SDK and works exclusively in the browser. It is not
  supported in mobile SDKs such as React Native, iOS, Android, or Flutter.
</Note>

Enabling debug mode in your browser can help troubleshoot issues with Forma. Here's how to activate it and what to look for in the logs.

### Activate Debug Mode

To enable debug mode, add `?formaDebug=true` to your app's URL (e.g., [`https://example.com?formaDebug=true)`](<https://example.com?formaDebug=true)>).&#x20;

#### View Debug Logs

1. Open your browser's developer tools:

- **Google Chrome/Edge**: Press `F12` or right-click and select "**Inspect" > "Console**".

- **Firefox**: Press `F12` or right-click and select "**Inspect Element" > "Console**".

- **Safari**: Press `Option + Command + C` to open developer tools and go to the "**Console**" tab.

#### Common Use Cases

Debug mode is helpful for:

- Verifying Forma initialization.

- Identifying issues with survey triggers.

- Troubleshooting unexpected behavior.

#### Debug Log Messages

Logs provide insights into:

- API calls and responses.

- Survey triggers and form interactions.

- Initialization errors.

If you're stuck, visit [**GitHub Discussions**](https://github.com/yldm-tech/forma/discussions) for assistance.
